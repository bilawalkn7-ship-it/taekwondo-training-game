import { techniqueCatalog } from './data/techniques.js';
import { ProgressSystem } from './game/progress/progressSystem.js';
import { TrainingSystem } from './game/training/trainingSystem.js';
import { createInitialGameState } from './game/logic/gameState.js';
import { UIManager } from './ui/uiManager.js';
import { PlayerController } from './game/player/playerController.js';
import { AIController } from './game/ai/aiController.js';
import { CombatSystem } from './game/combat/combatSystem.js';

class TaekwondoGameApp {
  constructor() {
    this.progress = new ProgressSystem();
    this.trainingSystem = new TrainingSystem(this.progress);
    this.gameState = createInitialGameState(this.progress.getProfile());
    this.ui = new UIManager(this);
    this.playerController = new PlayerController();
    this.aiController = new AIController();
    this.combatSystem = new CombatSystem();

    this.player = {
      x: 180,
      y: 60,
      width: 70,
      height: 120,
      health: 100,
      isBlocking: false,
      attackType: null,
      attackTimer: 0,
      attackCooldown: 0,
      vy: 0,
      vx: 0,
      facing: 1
    };

    this.enemy = {
      x: 700,
      y: 60,
      width: 70,
      height: 120,
      health: 100,
      isBlocking: false,
      attackType: null,
      attackTimer: 0,
      attackCooldown: 0,
      vy: 0,
      vx: 0,
      facing: -1
    };

    this.sparringActive = false;
    this.lastFrame = 0;
    this.renderLoop = this.renderLoop.bind(this);
    this.startMenu();
    requestAnimationFrame(this.renderLoop);
  }

  startMenu() {
    this.gameState.mode = 'menu';
    this.ui.render();
  }

  setMode(mode) {
    this.gameState.mode = mode;
    this.ui.render();

    if (mode === 'sparring') {
      this.startSparring();
    }
  }

  getSelectedTechnique() {
    return techniqueCatalog[this.gameState.selectedTechnique] || techniqueCatalog.frontKick;
  }

  startPracticeDrill() {
    const technique = this.getSelectedTechnique();
    this.gameState.feedback = `Practice: ${technique.name}. ${technique.prompt}`;
    this.gameState.objective = technique.prompt;
    this.gameState.practiceStep = 1;
    this.ui.render();
  }

  evaluatePracticeDrill() {
    const technique = this.getSelectedTechnique();
    const action = {
      guardUp: true,
      forwardStep: true,
      kneeChamber: true,
      kickExtension: true
    };

    const result = this.trainingSystem.evaluateFrontKick(action, this.gameState);

    this.gameState.performance.accuracy = result.score;
    this.gameState.performance.timing = 80;
    this.gameState.performance.reaction = 78;

    if (this.gameState.drillComplete) {
      this.progress.getProfile().trainingLevel = 'Yellow Belt';
      this.progress.saveProfile();
    }

    this.ui.render();
    return result;
  }

  startSparring() {
    this.sparringActive = true;
    this.gameState.mode = 'sparring';
    this.player.health = 100;
    this.enemy.health = 100;
    this.gameState.score = 0;
    this.ui.render();
  }

  stopSparring() {
    this.sparringActive = false;
    this.gameState.mode = 'menu';
    this.ui.render();
  }

  updateSparring(dt) {
    if (!this.sparringActive) return;

    const input = this.playerController.getMovement();

    if (input.left) this.player.x -= 200 * dt;
    if (input.right) this.player.x += 200 * dt;
    if (input.jump) this.player.vy = -420;
    this.player.x = Math.max(80, Math.min(400, this.player.x));

    this.player.vy += 900 * dt;
    this.player.y += this.player.vy * dt;
    if (this.player.y > 60) {
      this.player.y = 60;
      this.player.vy = 0;
    }

    this.enemy.x += this.aiController.state.mood === 'beginner' ? 0 : 0;
    this.aiController.update(this.enemy, this.player, dt);

    if (input.punch) {
      this.player.attackType = 'jab';
      this.player.attackTimer = 0.2;
      this.combatSystem.attemptAttack(this.player, this.enemy, 'jab');
    }

    if (input.kick) {
      this.player.attackType = 'kick';
      this.player.attackTimer = 0.35;
      this.combatSystem.attemptAttack(this.player, this.enemy, 'kick');
    }

    this.player.isBlocking = !!input.block;

    if (this.player.attackTimer > 0) {
      this.player.attackTimer -= dt;
      if (this.player.attackTimer <= 0) this.player.attackType = null;
    }

    if (this.enemy.attackType) {
      this.combatSystem.attemptAttack(this.enemy, this.player, this.enemy.attackType);
      this.enemy.attackType = null;
    }

    if (this.enemy.health <= 0 || this.player.health <= 0) {
      const result = this.enemy.health <= 0 ? 'win' : 'loss';
      this.progress.recordSparring(result);
      this.sparringActive = false;
      this.gameState.mode = 'menu';
      this.ui.render();
      return;
    }

    const playerBar = document.getElementById('player-health-bar');
    const enemyBar = document.getElementById('enemy-health-bar');
    if (playerBar) playerBar.style.width = `${this.player.health}%`;
    if (enemyBar) enemyBar.style.width = `${this.enemy.health}%`;
  }

  renderLoop(timestamp) {
    const dt = Math.min((timestamp - this.lastFrame) / 1000 || 0.016, 0.032);
    this.lastFrame = timestamp;

    this.updateSparring(dt);

    if (this.gameState.mode === 'sparring' && document.getElementById('sparring-arena')) {
      const playerNode = document.getElementById('player-fighter');
      const enemyNode = document.getElementById('enemy-fighter');
      if (playerNode) {
        playerNode.style.left = `${this.player.x}px`;
        playerNode.style.bottom = `${this.player.y + 60}px`;
      }
      if (enemyNode) {
        enemyNode.style.left = `${this.enemy.x}px`;
        enemyNode.style.bottom = `${this.enemy.y + 60}px`;
      }
    }

    requestAnimationFrame(this.renderLoop);
  }
}

const app = new TaekwondoGameApp();
window.app = app;
